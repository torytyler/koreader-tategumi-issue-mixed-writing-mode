+### Mixed writing modes: one writing mode per page
+
+Pages are laid out and painted in a single writing mode, so every writing-mode
+change (`DocFragment` with its own `writing-mode` and the block that carries it)
+must start a new page. Two upstream-neutral fixes protect that invariant; both
+are marked `FORK (tategumi)` and touch only small call sites:
+
+- `RN_SPLIT_MODE_CHANGE` (`crengine/include/lvpagesplitter.h`) marks a forced
+  page break that also changes the writing mode. The page splitter normally lets
+  an adjacent `page-break-inside: avoid` (e.g. an image line) suppress the
+  break; for a mode-changing break it now wins, otherwise the next fragment is
+  laid out on the previous page — in vertical-rl its text/illustration is then
+  painted over that page.
+- At a forced break the relevant margin is set to 0 (`lvrend.cpp`,
+  `pushVerticalMargin`), including negative collapsed margins. A negative
+  margin used to drag the next fragment a few pixels back, so its first line
+  started before the page boundary and belonged to the previous page. Fixing
+  the boundary also keeps illustrations from being drawn (with the wrong axis)
+  on the preceding text page.
+
+Same-mode image-only spine items keep sharing a page with text
+(`vertical_cover_page_spec.lua`); coverage for the mode-changing cases is in
+`spec/unit/vertical_mixed_mode_page_boundary_spec.lua`.
+
+Because these are layout changes, they must also invalidate cached page splits:
+`CRE_LAYOUT_ENGINE_REVISION` (`crengine/src/lvtinydom.cpp`, stored in the
+formerly-unused cache header padding field) is checked when a cache file is
+loaded, so caches written by an older engine are ignored (`CRE: ignoring cache
+file (layout engine revision mismatch)`) and the document is re-rendered. Bump
+it for any change that alters pagination; without that, readers would silently
+keep the page split they cached before the update.
+

+--[[--
+Regression: mixed-writing-mode pages must not share a page.
+
+Two EPUB authoring patterns used by Japanese vertical books broke the
+single-writing-mode-per-page assumption:
+
+1. A horizontal title page (its own style-h.css / writing-mode) followed by a
+   vertical body whose first element is a tiny "spacer" image
+   (`<p class="dummy"><img class="keep-space"/></p>`).  The spacer image line
+   is flagged `page-break-inside: avoid`; that "avoid" used to swallow the
+   forced page break carried by the DocFragment, so the vertical text was
+   laid out on the horizontal title page and painted over it.
+
+2. A vertical text page followed by an illustration page in horizontal mode.
+   The illustration's page break was correct, but the illustration's block
+   started a few pixels before the new page boundary, so the page range test
+   in DrawDocument kept it "visible" on the previous (vertical) page, where it
+   was painted with the wrong axis/scale on top of the text.
+
+Both end up as pages where text is unreadable and illustrations are drawn
+twice.  See doc/UPSTREAM_DIVERGENCE.md (mixed writing modes).
+
+Run via:
+  ./kodev test front -f "Vertical text: mixed-mode page boundaries"
+--]]
+
+local Blitbuffer = require("ffi/blitbuffer")
+
+describe("Vertical text: mixed-mode page boundaries", function()
+    local DocumentRegistry, ReaderUI, Screen, UIManager
+    local reader, path
+
+    setup(function()
+        require("commonrequire")
+        disable_plugins()
+        local device = require("device")
+        require("document/canvascontext"):init(device)
+        DocumentRegistry = require("document/documentregistry")
+        ReaderUI = require("apps/reader/readerui")
+        Screen = device.screen
+        UIManager = require("ui/uimanager")
+    end)
+
+    after_each(function()
+        if reader then reader:onClose(); reader = nil end
+        UIManager:quit()
+        if path then os.remove(path); path = nil end
+    end)
+
+    -- Build a small vertical EPUB. `docs` is an ordered list of
+    -- { name = "x.xhtml", content = "<body>...</body>", wm = "vertical-rl"|"horizontal-tb" }.
+    local function fixture(docs)
+        path = os.tmpname()
+        os.remove(path)
+        path = path .. ".epub"
+        local writer = require("ffi/archiver").Writer:new()
+        assert.is_true(writer:open(path, "zip"))
+        local function add(name, content)
+            assert.is_true(writer:addFileFromMemory(name, content))
+        end
+        add("mimetype", "application/epub+zip")
+        add("META-INF/container.xml", [[<container xmlns="urn:oasis:names:tc:opendocument:xmlns:container" version="1.0">
+<rootfiles><rootfile full-path="book.opf" media-type="application/oebps-package+xml"/></rootfiles></container>]])
+        local image = [[<svg xmlns="http://www.w3.org/2000/svg" width="560" height="700" viewBox="0 0 560 700">
+<rect width="560" height="700" fill="black"/></svg>]]
+        add("figure.svg", image)
+        -- Writing modes are carried by linked stylesheets (as publishers do
+        -- with style-v.css / style-h.css), not by a style= attribute: only then
+        -- does the fragment itself resolve to a concrete mode.
+        add("v.css", "html { writing-mode: vertical-rl }")
+        add("h.css", "html { writing-mode: horizontal-tb }")
+        local manifest, spine = {}, {}
+        for i, doc in ipairs(docs) do
+            add(doc.name, string.format([[<html xmlns="http://www.w3.org/1999/xhtml"><head>
+<link rel="stylesheet" type="text/css" href="%s"/>
+<style>
+body { margin: 0; padding: 0 } p { margin: 0 }
+img.keep-space { display: block; width: 0.1em; height: 0.1em; margin-right: -0.1em }
+.p-title { margin-top: 25%%; padding-left: 5%%; padding-right: 5%%; text-align: center; line-height: 1.6 }
+.p-title .book-title { margin: 0; padding-bottom: 2em; border-bottom: 1px solid #000000 }
+.p-title .author { margin: 0; padding-top: 1em }
+.p-caution { font-size: 0.9em; line-height: 1.6 }
+</style></head><body>%s</body></html>]],
+                doc.wm == "vertical-rl" and "v.css" or "h.css", doc.content))
+            manifest[#manifest + 1] = string.format(
+                '<item id="d%d" href="%s" media-type="application/xhtml+xml"/>', i, doc.name)
+            spine[#spine + 1] = string.format('<itemref idref="d%d"/>', i)
+        end
+        manifest[#manifest + 1] = '<item id="fig" href="figure.svg" media-type="image/svg+xml"/>'
+        manifest[#manifest + 1] = '<item id="v" href="v.css" media-type="text/css"/>'
+        manifest[#manifest + 1] = '<item id="h" href="h.css" media-type="text/css"/>'
+        add("book.opf", string.format([[<package xmlns="http://www.idpf.org/2007/opf" version="2.0" unique-identifier="id">
+<metadata xmlns:dc="http://purl.org/dc/elements/1.1/"><dc:identifier id="id">mixed-mode-page-boundary</dc:identifier>
+<dc:title>Mixed mode page boundary regression</dc:title><dc:language>ja</dc:language>
+<meta name="primary-writing-mode" content="vertical-rl"/></metadata>
+<manifest>%s</manifest><spine page-progression-direction="rtl">%s</spine></package>]],
+            table.concat(manifest), table.concat(spine)))
+        writer:close()
+        return path
+    end
+
+    local function vertical_paragraphs(prefix, count)
+        local out = {}
+        for i = 1, count do
+            out[#out + 1] = string.format("<p>%s%02d　縦書きの本文です。</p>", prefix, i)
+        end
+        return table.concat(out)
+    end
+
+    local function page_of(doc, text)
+        local hits = doc:findAllText(text, false, 0, 10, false)
+        assert.truthy(hits and hits[1], "missing text: " .. text)
+        return doc:getPageFromXPointer(hits[1].start)
+    end
+
+    -- Dark-pixel count of the current view of a page, restricted to the given
+    -- horizontal band (uses the actual UI draw, like the other ink-based specs).
+    local function ink_count(pageno, x_min, x_max)
+        reader.rolling:onGotoPage(pageno)
+        fastforward_ui_events()
+        local bb = Screen.bb
+        local dark = 0
+        for y = 5, Screen:getHeight() - 20, 3 do
+            for x = x_min, math.min(x_max, Screen:getWidth() - 1), 3 do
+                local px = bb:getPixel(x, y)
+                local r = type(px) == "number" and px or (px and px:getR())
+                if r and r < 0x80 then
+                    dark = dark + 1
+                end
+            end
+        end
+        return dark
+    end
+
+    it("keeps a vertical body off the preceding horizontal title page", function()
+        fixture({
+            {
+                name = "title.xhtml",
+                wm = "horizontal-tb",
+                -- The spacer image replicates the publisher dummy every spine
+                -- item of the reported book starts with.
+                content = '<p class="dummy"><img class="keep-space" src="figure.svg"/></p>'
+                    .. '<div class="p-title"><div class="book-title"><p>TITLEPAGE</p></div>'
+                    .. '<div class="author"><p>SUBTITLE LINE</p></div>'
+                    .. "<p><br/></p><p><br/></p><p><br/></p>"
+                    .. '<div class="label"><p>PUBLISHER LABEL</p></div></div>',
+            },
+            {
+                name = "body.xhtml",
+                wm = "vertical-rl",
+                content = '<p class="dummy"><img class="keep-space" src="figure.svg"/></p>'
+                    .. '<div class="p-caution">' .. vertical_paragraphs("本文", 40) .. "</div>",
+            },
+        })
+        reader = ReaderUI:new{
+            dimen = Screen:getSize(),
+            document = DocumentRegistry:openDocument(path),
+        }
+        UIManager:show(reader)
+        fastforward_ui_events()
+        local doc = reader.document
+        local title_page = page_of(doc, "TITLEPAGE")
+        local body_page = page_of(doc, "本文01")
+        print(string.format("[mixed_page] title=%d body=%d pages=%d modes=%s/%s/%s", title_page, body_page,
+            doc:getPageCount(), doc._document:getPageWritingMode(1), doc._document:getPageWritingMode(2),
+            doc._document:getPageWritingMode(3)))
+        assert.is_true(body_page > title_page,
+            "vertical body was laid out on the horizontal title page")
+    end)
+
+    it("does not paint a horizontal illustration page over the preceding vertical text page", function()
+        fixture({
+            { name = "text.xhtml", wm = "vertical-rl", content = vertical_paragraphs("直前", 6) },
+            {
+                name = "figure.xhtml",
+                wm = "horizontal-tb",
+                content = '<p class="dummy"><img class="keep-space" src="figure.svg"/></p>'
+                    .. '<p><img src="figure.svg" width="560" height="700"/></p>',
+            },
+            { name = "after.xhtml", wm = "vertical-rl", content = "<p>AFTERFIGURE</p>" },
+        })
+        reader = ReaderUI:new{
+            dimen = Screen:getSize(),
+            document = DocumentRegistry:openDocument(path),
+        }
+        UIManager:show(reader)
+        fastforward_ui_events()
+        local doc = reader.document
+
+        local text_page = page_of(doc, "直前06")
+        local figure_page = doc:getPageFromXPointer("/body[1]/DocFragment[2]/body[1]/p[1]/img[1].0")
+        print(string.format("[mixed_page] text=%d figure=%d", text_page, figure_page))
+        assert.is_true(figure_page > text_page,
+            "illustration was not given its own page")
+
+        -- Vertical columns sit on the right of the page: the left half of the
+        -- text page must be (nearly) blank. A leaked, wrong-axis illustration
+        -- paints a full-page black rectangle there.
+        local left_ink = ink_count(text_page, 0, math.floor(Screen:getWidth() * 0.35))
+        print(string.format("[mixed_page] text page left-third ink=%d", left_ink))
+        assert.is_true(left_ink < 40,
+            string.format("illustration was painted over the text page (left-third ink=%d)", left_ink))
+    end)
+end)
